# LBが気にするべき範囲のLongPacketのフォーマット
format LongPacketHeaderForLB:
   form :u1 # 1 long header
   fixed :u1 # 1だがgrease拡張などで動かされている可能性あり
   long_packet_type :u2 # non encrypted
   reserved :u2 # encrypted
   packet_number_length :u2 # encrypted
   version :u32 # non encrypted
   dst_conn_id_len :u8
   dst_conn_id :[dst_conn_id_len]u8
   remain :[..]u8 # omitted

# LBが気にするべき範囲のShortPacketのフォーマット
format ShortPacketHeaderForLB:
    from :u1 # 0 short header
    fixed :u1 # 1だがgrease拡張などで動かされている可能性あり
    spin :u1 # non encrypted
    reserved :u2 # encrypted
    key_phase :u1 # encrypted
    packet_number_length :u2 # encrypted
    dst_conn_id :[..]u8
    remain :[..]u8 # omitted


format FirstByteOfQUICLB:
  config_rotation :u3
  cid_len_or_random :u5

format QUICLBConnectionID:
  first_octet :FirstByteOfQUICLB
  server_id :[1..]u8 # ここは実装で決め打ち
  nonce :[4..]u8 # ここも実装で決め打ち

format QUICLBIDForThisImplementation: # とりあえずこの構造で仮決定 (20バイト)
  first_octet :FirstByteOfQUICLB # cid_len_or_randomはランダムビット
  server_id :u4 # server id (今回は通常は2台だし増やすとしても16台も使うことはないのでこれでよし)
  client_id_high :u4
  client_id_low :[2]u8 # also nonce for each connection id (2^20=1048576~=約100万)
  nonce_pad :u7 # nonce要件を満たすためのランダム値
  mac_high :u1
  mac_low :[15]u8 # mac calucluated with connection id exclude this field and server/lb shared secret 
              # ただしパケット長とのトレードオフ 
              # あとrandomビット数が5bit + 20(=4+2*8)bit + 7bit = 32bitでQUIC-LB draftのnonce要件32bitを満たす
              # quic-goもといRFC9000により長さは20バイトが限界であると判明
              # mac計算時にはQUICLBIDFortThisImplementationMACTargetとshared secretを使って128bit MACを計算し
              # そのうちの最上位バイトの下位1ビット及び残りの15バイトをMAC(121bit)とする

format QUICLBIDFortThisImplementationMACTarget:
  first_octet :FirstByteOfQUICLB
  server_id :u4
  client_id_high :u4
  client_id_low :[2]u8 
  nonce_pad :u7
  zero :u1 # paddingとして0を埋める


format QUICLBIDForThisImplementationv2: # 
  first_octet :FirstByteOfQUICLB # cid_len_or_randomはランダムビット
  server_id :u4 # server id (今回は通常は2台だし増やすとしても16台も使うことはないのでこれでよし)
  opaque_1 :u4
  opaque :[18]u8

# 'input_bytes' is drawn from one half of the plaintext.  It forms
# the N most significant octets of the output, where N is half the
# 'length' argument, rounded up, and thus a number between 3 and 10,
#  inclusive.
# 'Zeropad' is a set of 14-N octets set to zero.
# 'length' is an 8-bit integer that reports the sum of the
# configured nonce length and server id length in octets, and forms
# the fifteenth octet of the output.  The 'length' argument MUST NOT
# exceed 19 and MUST NOT be less than 5.
# 'pass' is an 8-bit integer that reports the 'pass' argument of the
# algorithm, and forms the sixteenth (least significant) octet of
# the output.  It guarantees that the cryptographic input of every
# pass of the algorithm is unique.
format ExpandResult:
  input_bytes :[3..=10]u8
  zero_padding :[14 - input_bytes.length]u8
  full_connection_id_len :u8
  full_connection_id_len >= 5 && full_connection_id_len <= 19
  pass :u8 # for cryptographic input uniqueness

- Initial/Handshake/Zero RTT/Retry
  - first_octtet & 0x80 == 1 -> LongPacket
- One RTT
  - first_octet & 0x40 == 1 -> ShortPacket

LongPacket
  + input + 5 == dst_conn_id_len
  + input + 6 ~ input + 6 + dst_conn_id_len == dst_conn_id
ShortPacket 
  + input + 1 == dst_conn_id (length is defined by implementation,LBの場合cid_len_or_randomでわかるはず)
